cmake_minimum_required(VERSION 3.16)
project(mid360_lio_frame_probe)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(Eigen3 REQUIRED NO_MODULE)
find_package(tf2_ros REQUIRED)
find_package(tf2_eigen REQUIRED)
find_package(PCL REQUIRED)
find_package(OpenMP REQUIRED)
get_filename_component(AEROMIND_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(LIO_SOURCE "${AEROMIND_ROOT}/src/third_party/spark-fast-lio/spark_fast_lio")
add_executable(mid360_lio_frame_probe main.cpp)
target_compile_features(mid360_lio_frame_probe PRIVATE cxx_std_20)
# Test only: call actual private output API without changing the production class.
target_compile_options(mid360_lio_frame_probe PRIVATE -fno-access-control)
target_compile_definitions(mid360_lio_frame_probe PRIVATE ROOT_DIR="${LIO_SOURCE}/")
target_include_directories(mid360_lio_frame_probe PRIVATE "${LIO_SOURCE}/include" "${LIO_SOURCE}/third_party/IKFoM_toolkit" "${LIO_SOURCE}/third_party/ikd-Tree" ${PCL_INCLUDE_DIRS})
ament_target_dependencies(mid360_lio_frame_probe rclcpp sensor_msgs nav_msgs geometry_msgs pcl_conversions tf2_ros tf2_eigen)
target_link_libraries(mid360_lio_frame_probe "${AEROMIND_ROOT}/install_mid360/spark_fast_lio/lib/libspark_lio_component.so" ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)
