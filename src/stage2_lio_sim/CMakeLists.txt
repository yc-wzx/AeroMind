cmake_minimum_required(VERSION 3.8)
project(stage2_lio_sim)
find_package(ament_cmake REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
find_package(ignition-sensors6 REQUIRED COMPONENTS imu)
add_library(stage2_kinematic_imu SHARED kinematic_imu.cpp)
target_compile_features(stage2_kinematic_imu PUBLIC cxx_std_17)
target_link_libraries(stage2_kinematic_imu ignition-gazebo6::ignition-gazebo6
 ignition-plugin1::register ignition-sensors6::imu)
target_link_options(stage2_kinematic_imu PRIVATE "-Wl,-z,defs")
install(TARGETS stage2_kinematic_imu DESTINATION lib)
install(PROGRAMS scripts/lio_sensor_adapter.py scripts/lio_odometry_adapter.py
 scripts/lio_navigation_interface.py scripts/lio_contract.py DESTINATION lib/${PROJECT_NAME})
ament_package()
