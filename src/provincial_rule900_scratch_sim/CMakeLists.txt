cmake_minimum_required(VERSION 3.8)
project(provincial_rule900_scratch_sim)
find_package(ament_cmake REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-plugin1 REQUIRED COMPONENTS register)
find_package(ignition-sensors6 REQUIRED COMPONENTS imu)
add_library(inverted_position_imu SHARED plugins/inverted_position_imu.cpp)
add_library(stage2_gimbal_joint SHARED plugins/gimbal_joint.cpp)
foreach(target inverted_position_imu stage2_gimbal_joint)
  target_compile_features(${target} PUBLIC cxx_std_17)
  target_link_libraries(${target} ignition-gazebo6::ignition-gazebo6 ignition-plugin1::register ignition-sensors6::imu)
  target_link_options(${target} PRIVATE "-Wl,-z,defs")
endforeach()
install(TARGETS inverted_position_imu stage2_gimbal_joint DESTINATION lib)
set(nodes circle_navigation radar_fixed_odometry radar_host_command radar_sim_electrical inverted_sim_format_adapter gimbal_lio_mapping ego_trajectory_executor ego_goal_adapter)
file(GLOB helpers CONFIGURE_DEPENDS scripts/*.py)
foreach(node ${nodes})
  install(PROGRAMS scripts/${node}.py DESTINATION lib/${PROJECT_NAME})
  list(REMOVE_ITEM helpers ${CMAKE_CURRENT_SOURCE_DIR}/scripts/${node}.py)
endforeach()
install(FILES ${helpers} scripts/circle_contract.json DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch worlds config DESTINATION share/${PROJECT_NAME})
ament_package()
